---@class ColorScheme
<* for name, value in colors *>---@field {{ name | camel_case }} string
<* endfor *>

---@class Theme 
---@field light ColorScheme
---@field dark ColorScheme

---@type Theme 
local Theme = {
	light = {
		<* for name, value in colors *>{{ name | camel_case }} = "0xff{{ value.light.hex_stripped }}",
		<* endfor *>},
	dark = {
		<* for name, value in colors *>{{ name | camel_case }} = "0xff{{ value.dark.hex_stripped }}",
		<* endfor *>},
}

return Theme
